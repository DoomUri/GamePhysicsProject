    <script type="text/javascript">
	C = 0;
	D = 0;
	function Parar(){
		clearInterval(Interval);
		document.getElementById("Bola").style.left=10+"px";
		document.getElementById("Bola").style.bottom=200+"px";
		D = 0;
	}
	function Començar(){
 		if(D == 0){
			t = 0;
			Interval = setInterval(Moviment, 10);
			D = 1;
		}
	}
	function Moviment(){
		v0 = document.getElementById("Velocitat").value;	
		B = document.getElementById("Angle").value;
		A = (B*2*Math.PI)/(360);
		v0x = v0*Math.cos(A);
		v0y = v0*Math.sin(A);
		a = document.getElementById("Gravetat").value;
		x = 10 + v0x*t;
		document.getElementById("Bola").style.left = x + "px";
		y = 200 + v0y*t -0.5*a*t*t;
		document.getElementById("Bola").style.bottom = y + "px";
		t = t + 0.01;
	}
	function Trajectoria(){
		if(C == 0){
			for(i=0;i<100;i++){
				document.getElementById("Contenidor").innerHTML += "<div id='Bola"+i+"' class='Duplicats'></div>";
				v0 = document.getElementById("Velocitat").value;	
				B = document.getElementById("Angle").value;
				A = (B*2*Math.PI)/(360);
				v0x = v0*Math.cos(A);
				v0y = v0*Math.sin(A);
				a = document.getElementById("Gravetat").value;
				x = 10 + v0x*i;
				document.getElementById("Bola"+i).style.left = x + "px";
				y = 200 + v0y*i - 0.5*a*i*i;
				document.getElementById("Bola"+i).style.bottom = y + "px";
			}
		}
		i = 0;
		C = 1;
	}
	function Esborrar(){
		if(C == 1){
			for(i=0;i<100;i++){
				element = document.getElementById("Bola"+i);
				element.remove();
			}
		}
		i = 0;
		C = 0;
	}
