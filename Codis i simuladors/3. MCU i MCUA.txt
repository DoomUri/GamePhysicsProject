    <script type="text/javascript">
	D = 0;
	C = 0;
	function Començar(){
		if(D == 0){
			r = document.getElementById("radi").value; //radi
			v0 = document.getElementById("v0").value; //velocitat angular inicial en graus/s
			a = document.getElementById("a").value; //acceleració angular en graus/s^2
			w0 = (v0*2*Math.PI)/360; //velocitat angular inicial en rad/s
			k = (a*2*Math.PI)/360; //acceleració angular en rad/s^2
			t = 0;
			Interval = setInterval(Moviment, 10);
			D = 1;
		}
	}
	function Moviment(){
		g = w0*t + 0.5*k*t*t;
		x = 490 + r*Math.cos(g);
		y = 340 + r*Math.sin(g);
		document.getElementById("Bola").style.left = x + "px";
		document.getElementById("Bola").style.bottom = y + "px";
		t = t + 0.01;
	}
	function Parar(){
		clearInterval(Interval);
		D = 0;
	}
	function Trajectoria(){
		if(C == 0){
			r = document.getElementById("radi").value; //radi
			for(i=0;i<30;i++){
				document.getElementById("centre").innerHTML += "<div id='Bola"+i+"' class='grajo'></div>"
				h = (i*12*2*Math.PI)/360;
				x0 = r*Math.cos(h);
				y0 = r*Math.sin(h);
				document.getElementById("Bola"+i).style.left = x0 + "px";
				document.getElementById("Bola"+i).style.bottom = y0 + "px";
			}
			C = 1;
		}
	}
	function Esborrar(){
		for(i=0;i<30;i++){
			element = document.getElementById("Bola"+i);
			element.remove();
		}
		C = 0;
	}
    </script>
